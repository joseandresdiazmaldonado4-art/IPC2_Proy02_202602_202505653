Inicialización mediante XML

Las categorías usan su contenido como nombre y pueden indicar un atributo padre. La carga conserva los datos existentes.

<config>
  <listaCategorias>
    <categoria>Ciencia</categoria>
    <categoria padre="Ciencia">Física</categoria>
  </listaCategorias>
  <listaLibros>
    <libro><ISBN>978001</ISBN><titulo>Fundamentos</titulo><autor>Ana López</autor><categoria>Física</categoria></libro>
  </listaLibros>
</config>
